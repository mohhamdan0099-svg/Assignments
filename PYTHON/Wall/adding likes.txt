class Message(models.Model):
    message = models.TextField()
    # This creates a Many-to-Many relationship between User and Message
    # related_name allows you to do message.user_likes.all()
    user_likes = models.ManyToManyField(User, related_name="liked_messages")
    user = models.ForeignKey(User, related_name="messages", on_delete=models.CASCADE)
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)



def like_message(request, message_id):
    if 'user_id' not in request.session:
        return redirect('/')
    
    # Get the current logged-in user and the target message
    user = User.objects.get(id=request.session['user_id'])
    message = Message.objects.get(id=message_id)
    
    # Check if the user already liked this message
    if user in message.user_likes.all():
        # Unlike: remove user from likes
        message.user_likes.remove(user)
    else:
        # Like: add user to likes
        message.user_likes.add(user)
        
    return redirect('/wall')


path('wall/like/<int:message_id>', views.like_message),
